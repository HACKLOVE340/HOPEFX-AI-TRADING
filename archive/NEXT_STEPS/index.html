# Create first test

import pytest
from strategies import MovingAverageCrossover

def test_strategy_creation():
    strategy = MovingAverageCrossover(
        symbol="EUR/USD",
        timeframe="1h",
        fast_period=10,
        slow_period=30
    )
    assert strategy.symbol == "EUR/USD"
    assert strategy.fast_period == 10
EOF

# Run tests
pytest tests/
